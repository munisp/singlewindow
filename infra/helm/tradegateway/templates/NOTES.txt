1. Check all pods are running:
   kubectl get pods -n {{ .Values.global.namespace }}

2. Verify health endpoints:
   kubectl exec -n {{ .Values.global.namespace }} deploy/declaration-service -- \
     wget -qO- http://localhost:8080/health

3. Check TigerBeetle mode (must be 'live' in production):
   kubectl exec -n {{ .Values.global.namespace }} deploy/tigerbeetle-bridge -- \
     wget -qO- http://localhost:9100/health | jq .mode

4. Verify Temporal worker connected:
   kubectl logs -n {{ .Values.global.namespace }} deploy/temporal-worker | \
     grep "Worker started"

5. Check Prometheus targets:
   kubectl port-forward -n monitoring svc/prometheus 9090:9090 &
   open http://localhost:9090/targets
